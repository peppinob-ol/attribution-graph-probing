\section{Pipeline Reference}
\label{appx:pipeline}

\subsection{Pseudocode}
\begin{small}
\begin{verbatim}
Input: attribution_graph G_p, seed_prompt p, target_logit y
1. V_tau := select_nodes_by_cumulative_influence(G_p, tau=0.95)
2. Concepts := llm_generate_concepts(p, K)
3. Probes := synthesize_probes(Concepts, template_constraints)
4. for f in V_tau:
     for q in Probes:
       A[f, q] := measure_activation(f, q)        # Neuronpedia API or local
5. CPAS := aggregate_cross_prompt(A)              # Sec 3.2
6. for f in V_tau:
     role[f] := classify(CPAS[f], thresholds)     # Sec 3.4 strict-priority rules
     name[f] := assign_name(f, A, blacklist, target_token_rules)
7. Supernodes := group_by(role, name) with stability >= 0.6
8. Subgraph := pin_nodes(Supernodes, embeddings, y)
9. for (e_A, e_B) in pairs:
     I_lab  := build_swap(S(e_A), S(e_B), M_a=-2, M_b=20)
     I_rand := matched_random_control(I_lab)       # Def 5
     run_swap(I_lab); run_swap(I_rand)             # freeze_attention=false
10. Adaptive_M_search on missed pairs              # Sec 3.7
\end{verbatim}
\end{small}

\subsection{CPAS metrics}
\begin{itemize}\itemsep0pt
\item \texttt{peak\_consistency\_main}: fraction of probes whose peak token matches the modal peak.
\item \texttt{n\_distinct\_peaks}: number of distinct peak tokens across active probes.
\item \texttt{share\_F} / \texttt{conf\_F}: fraction of active probes whose peak is on a functional token / classifier confidence on that share.
\item \texttt{func\_vs\_sem\_pct}: $100 \cdot (\max\mathrm{func} - \max\mathrm{sem})/\max\mathrm{overall}$.
\item \texttt{sparsity\_median}: median of $(\mathrm{peak} - \mathrm{mean})/\mathrm{peak}$ across active probes.
\item \texttt{conf\_S}: $1 - \mathrm{share\_F}$.
\end{itemize}

\subsection{Decision-rule thresholds (frozen pre-analysis)}
\textsc{Sem-Dict} \texttt{peak\_consistency}$\geq 0.80$ \textsc{and} \texttt{n\_distinct\_peaks}$\leq 1$.
\textsc{Say-X} \texttt{func\_vs\_sem\_pct}$\geq 50$ \textsc{and} \texttt{conf\_F}$\geq 0.90$ \textsc{and} layer $\geq 7$.
\textsc{Rel} \texttt{sparsity\_median}$<0.45$.
\textsc{Sem-Conc} layer $\leq 3$ \textsc{or} \texttt{conf\_S}$\geq 0.50$ \textsc{or} \texttt{func\_vs\_sem\_pct}$<50$.
Otherwise \textsc{Review} (excluded).

\textbf{Conjunctions:} \texttt{and, or, but, nor, so, yet, for}.

\textbf{Auxiliaries:} \texttt{do, does, did, have, has, had, will, would, shall, should, can, could, may, might, must}.

\paragraph{Directionality.}
\emph{Forward:} copulas, auxiliaries, articles, most prepositions. \emph{Backward:} \texttt{of}, possessive \texttt{'s}. \emph{Bidirectional:} conjunctions, punctuation. Window: 7 tokens. Multi-token entities resolve to first subword.

\paragraph{Documented French failure mode.}
Applying the English pipeline to ``le contraire de `petit' est'' with English probes produced incoherent activations and no valid supernode grouping. After adding French probes and expanding the blacklist (mixed FR/EN bridging tokens), grouping became reliable; we report this as a known cross-lingual limitation.

\section{Adaptive M-Search}
\label{appx:msearch}

\subsection{Two-phase protocol}
\textbf{Phase 1 (coarse, geometric):} $M \in \{0.1, 0.25, 0.6, 1.5, 4.0, 10.0, 20.0\}$ probed in order; stop at first Hit. \textbf{Phase 2 (KL-binary):} compute KL$(M)$ at the Phase-1 grid; identify the steepest KL transition; binary-refine $M$ in that interval for 6 steps. Accept hit at any refinement point.

\subsection{Winning-$M$ histogram}
93\% of hits found in Phase 1; bimodal at $M{\sim}2.4$ (47\%) and $M{\sim}6.9$ (40\%); long tail to $M{\geq}10$ (13\%).

\subsection{High-$M$ negative result}
$M \in \{50, 100, 200\}$ on near-miss pairs across 3 domains: 80 runs, \emph{0 new hits}; existing hits at $M{=}20$ degraded monotonically with increasing $M$.

\subsection{Top-$k$ negative result}
$k \in \{1, 3, 5, 10\}$ by graph influence at $M \in [20, 200]$ on Products: 96 runs, \emph{0 hits}. Only the full feature set ($k{=}67$) achieves hits. Graph influence correlates negatively with stored activation ($\rho{=}-0.362$, $p{=}0.003$).

\subsection{Five failure modes}
(1) \emph{Severe overshoot} (rescuable by $M$ reduction; 88\% rescue rate; 37/42 \texttt{oklahoma\_tulsa} pairs at $M{=}5$).
(2) \emph{Field interference} (rescuable by mid$+$answer subset; \cref{appx:fieldadd}).
(3) \emph{Feature interaction noise} (garbage at high $M$ despite positive rank improvement; \texttt{indiana$\to$arkansas}: \texttt{tonode} at $M{=}20$ $\to$ \texttt{Little Rock} at $M{=}5$).
(4) \emph{Signal collapse} (vsMax collapses below pair-specific $M$; \texttt{vermont$\to$kansas}: vsMax $8.3$ at $M{=}20$ $\to$ $0.3$ at $M{=}5$).
(5) \emph{Feature specificity} (correct context, wrong answer; \texttt{Hutchinson} not \texttt{Topeka}; not fixable by $M$).

\section{KL Divergence as a Steering Diagnostic}
\label{appx:kl}

\paragraph{Linearity with $M$.}
Per-pair KL$(M)$ is highly linear (R$^2 > 0.93$ for 8/10 USA pairs; range 0.85--1.00). Mean KL by $M$: $M{=}5$ $\to$ 9.33; $M{=}10$ $\to$ 11.10; $M{=}20$ $\to$ 13.03.

\paragraph{KL$\geq 12$ as a hit-veto.}
Across 30 USA observations, no hit occurs at KL$\geq 12$. As a binary classifier on $\{$KL$<12\}$: recall 100\% (7/7), precision 37\% (7/19; 12 false positives split among feature-specificity (3), evaluator gaps (3), signal collapse (3), near-threshold (3)).

\paragraph{KL slope vs intervention structure.}
$r(\mathrm{slope}, \mathrm{ablate\_count}) = +0.68$; $r(\mathrm{intercept}, \mathrm{ablate\_count}) = -0.63$. More source features ablated $\to$ steeper KL increase per unit $M$. High-slope pairs are most sensitive to $M$ and most likely to benefit from $M$-reduction.

\paragraph{Use in adaptive search.}
(i) Run at $M{=}20$ and measure KL at pos 0; (ii) if KL$\geq 12$, fit KL$(M)$ from $\geq 1$ extra point and compute $M_{\mathrm{crit}} = (12 - b)/a$; (iii) re-run at $M = \lfloor 0.8 \cdot M_{\mathrm{crit}} \rfloor$; (iv) if intercept $b\geq 12$, flag as intrinsically disruptive.

\end{itemize}

\subsection{Suppression-only zone (source absent, no target)}
USA 59.2\%, Books 48.3\%, Products 46.2\%, Paintings 68.9\%, Sounds 66.7\%. The single largest outcome category in every domain: suppression is generic; targeting requires specific features.

\subsection{Threats to validity for case studies}
Selection bias toward extremes; tier unavailability for the best-FA variant at initial analysis; sounds shared-answer artefacts (12/30 pairs); single-example interpretations should not be over-generalized; explanations are post-hoc and constitute hypotheses, not conclusions.

\section{Sounds and Paintings Deep Dive}
\label{appx:negative}

\paragraph{Why Sounds fails Condition~2.}
Three structural causes converge: (i) $N{=}6$ entities; (ii) $12/30$ non-identity pairs share the answer ``brown'', so any source-suppression in those pairs incidentally counts as a target-hit for random; (iii) $K{=}6$ colors with near-uniform $\sim 0.15$ probability mass make the logit margin too thin to overshoot precisely. The vsMax gap is $+0.14$ (Negligible); position-0 vsMax actually \emph{favors} random ($+0.30$ vs labeled $+0.68$ in median terms; \cref{appx:pos0}).

\paragraph{Why Paintings is weak.}
The \texttt{first\_name} answer field is fundamentally coarse: \texttt{Claude}, \texttt{Pablo}, \texttt{Leonardo} are common tokens with many non-painting associations. The \texttt{painter} supernode subsumes \texttt{first\_name} (since ``Monet'' contains information about ``Claude''), making the two fields effectively identical for matching. With 17\% error-node rates (highest of any domain), much of the circuit is invisible. 89/90 pairs fail at the \texttt{painter+first\_name} variant; the only hit (Grande Jatte $\to$ Water Lilies) requires baseline rank 3.

\paragraph{Recommendation.}
We exclude Sounds from cross-domain inference for C1 and treat Paintings as a Weak-strength supporting domain rather than a confirmatory one. The four-domain rank-perfect scaffold prediction (\cref{appx:scaffold}) holds without Sounds.

\paragraph{Runtime profile.}
Per single-circuit pipeline (graph $+$ activation $+$ grouping $+$ subgraph): 14--24 minutes on a single L4 GPU. Full per-domain swap sweeps: USA 17h, Books 1.6h, Products 0.9h, Paintings 0.6h, Sounds 0.2h on 8 H100 GPUs (matched-control replicates parallelized).

For each domain: (a) attribution graphs (Neuronpedia JSON); (b) probe-prompt files; (c) activation matrices; (d) supernode groupings; (e) swap dumps with per-pair Hit/vsMax/RkGrp/Sup/Flip/CtrlS/KL trajectories; (f) field-additivity per-variant tables; (g) adaptive-$M$ logs (Phase 1 \& Phase 2); (h) random-control sha256 seeds.

\section{Glossary and Notation}
\label{appx:glossary}

\textbf{CLT}: Cross-Layer Transcoder.
\textbf{CPAS}: Cross-Prompt Activation Signature.
\textbf{$\tau$}: cumulative-influence threshold for feature selection (default $0.95$).
\textbf{$M_{\mathrm{ablate}}$, $M_{\mathrm{amplify}}$}: multipliers for source/target features (defaults $-2$, $20$).
\textbf{vsMax}: best$_t\big(\ell_{e_B}(t) - \max_{e'\neq e_B} \ell_{e'}(t)\big)$.
\textbf{RkGrp}: best target rank within domain answer set.
\textbf{Target Recovery}: target logit beats its own unsteered baseline at any trajectory position.
\textbf{Regime A/C/D/E}: position-0 logit-shift regimes (\cref{appx:regime}).
\textbf{Scaffold}: features shared between source/target graphs with identical supernode assignment.
\textbf{Scaffold influence}: fraction of total graph influence carried by scaffold features.
\textbf{KL}: KL$(\mathrm{baseline} \| \mathrm{steered})$ at position 0.
